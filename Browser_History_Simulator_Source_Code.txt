// ==========================================
// STACK IMPLEMENTATION (FROM SCRATCH)
// ==========================================
class Stack {
  constructor() {
    this.items = [];
  }

  // Add element to top of stack
  push(element) {
    this.items.push(element);
  }

  // Remove and return top element
  pop() {
    if (this.isEmpty()) return null;
    return this.items.pop();
  }

  // Return top element without removing
  peek() {
    if (this.isEmpty()) return null;
    return this.items[this.items.length - 1];
  }

  // Check if stack is empty
  isEmpty() {
    return this.items.length === 0;
  }

  // Return number of elements
  size() {
    return this.items.length;
  }

  // Remove all elements
  clear() {
    this.items = [];
  }

  // Return items (top first) for display
  getItems() {
    return [...this.items].reverse();
  }
}

// ==========================================
// GLOBAL STATE
// ==========================================
const backStack = new Stack();
const forwardStack = new Stack();
let currentPage = "about:blank";

// ==========================================
// CORE FUNCTIONS
// ==========================================

// Visit a new page
function visit() {
  let url = document.getElementById("urlInput").value.trim().toLowerCase();

  if (url === "") {
    alert("Please enter a URL");
    return;
  }

  // Clean URL
  url = url.replace(/^https?:\/\//, "").replace(/\/$/, "");

  if (url === currentPage) {
    return;
  }

  // Push current page to Back Stack
  if (currentPage !== "about:blank") {
    backStack.push(currentPage);
  }

  // Clear Forward Stack (important browser behavior)
  forwardStack.clear();

  // Set new current page
  currentPage = url;
  document.getElementById("urlInput").value = "";
  updateUI();
}

// Go Back
function goBack() {
  if (backStack.isEmpty()) return;

  // Save current page to Forward Stack
  forwardStack.push(currentPage);

  // Move to previous page
  currentPage = backStack.pop();
  updateUI();
}

// Go Forward
function goForward() {
  if (forwardStack.isEmpty()) return;

  // Save current page to Back Stack
  backStack.push(currentPage);

  // Move to next page
  currentPage = forwardStack.pop();
  updateUI();
}

// Home / Reset
function goHome() {
  backStack.clear();
  forwardStack.clear();
  currentPage = "about:blank";
  document.getElementById("urlInput").value = "";
  updateUI();
}

// ==========================================
// UI UPDATE
// ==========================================
function updateUI() {
  document.getElementById("currentPage").textContent = currentPage;

  if (currentPage === "about:blank") {
    document.getElementById("pageInfo").textContent = "Enter a URL and click Visit";
  } else {
    document.getElementById("pageInfo").textContent = "Currently viewing: " + currentPage;
  }

  // Enable/Disable buttons
  document.getElementById("btnBack").disabled = backStack.isEmpty();
  document.getElementById("btnForward").disabled = forwardStack.isEmpty();

  // Render stacks
  renderStack("backDisplay", backStack);
  renderStack("forwardDisplay", forwardStack);
}

function renderStack(id, stack) {
  const container = document.getElementById(id);
  const items = stack.getItems();

  if (items.length === 0) {
    container.innerHTML = "Empty";
    return;
  }

  container.innerHTML = items.map(function(url, index) {
    return '<div class="item">' + url + (index === 0 ? " (TOP)" : "") + '</div>';
  }).join("");
}

// Initialize
updateUI();
</script>

CORE LOGIC SUMMARY
================================================================================

Stack Class Methods:
- push(element)  → Insert at top
- pop()          → Remove from top
- peek()         → View top element
- isEmpty()      → Check if empty
- clear()        → Empty the stack

Main Operations:
- Visit   → Push current to Back Stack + Clear Forward Stack
- Back    → Push current to Forward Stack + Pop from Back Stack
- Forward → Push current to Back Stack + Pop from Forward Stack
- Home    → Clear both stacks
